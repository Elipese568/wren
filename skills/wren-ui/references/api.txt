Wren API（按头文件整理，v0.2.1）
================================================================
命名空间 wren。所有坐标是 DIP；所有颜色是 Color。签名省略了 std:: 前缀以外的默认参数。

struct FrameStats {
    unsigned long long frames;
    double render_ms;      // 形状 + 文字（CPU）平均耗时
    double paint_ms;       // 形状/清屏/布局
    double text_ms;        // DirectWrite 文字
    double present_ms;     // 上屏
    double interval_ms;    // 相邻帧起点间隔中位数
    double interval_p95_ms;
    double interval_max_ms;
    int    dropped;
    int    refresh_hz;
};

struct HostOptions {
    std::wstring title = L"Wren";
    std::wstring window_class = L"WrenHostWindow";
    int width_dip = 720;
    int height_dip = 480;
    Appearance appearance = Appearance::Auto;
    MaterialPreference material = MaterialPreference::Auto;
    std::function<void()> on_escape;   // 为空时 Esc 关窗
};

class Host {
    bool create(HINSTANCE, std::unique_ptr<Widget> root, HostOptions = {});
    int  run();
    void quit(int exit_code = 0);
    void close();

    HWND hwnd() const;
    Widget* root() const;

    const Theme& theme() const;
    Appearance appearance() const;
    void set_appearance(Appearance);
    MaterialKind material_kind() const;
    MaterialPreference material_preference() const;
    void set_material(MaterialPreference);
    const wchar_t* presenter_name() const;
    PresenterKind presenter_kind() const;

    FrameStats frame_stats() const;
    void reset_frame_stats();

    float dpi() const;
    float pixels(float dip) const;
    float dip(float px) const;
    int client_width_px() const;
    int client_height_px() const;

    void invalidate(const RectF* dirty = nullptr);   // 当前实现全量重画
    void layout_changed();
    void set_tick_handler(std::function<void()>);
    void set_animating(bool animating, int interval_ms = 33);  // <=0 跟随合成器时钟
    void schedule_once(int delay_ms, std::function<void()>);

    Widget* focused_widget() const;
    void focus_widget(Widget*);
    bool focus_next(bool backwards);

    // MSAA（automation.cpp 用）
    struct AccessibleEntry { Widget* widget; int row; };
    std::vector<AccessibleEntry> accessible_entries() const;
    int  child_id_of(const Widget* widget) const;
    void notify_accessible(const Widget* widget, LONG event_id);
    void notify_list_selection(const Widget* table, int row);
    RectF to_screen(const RectF& client_dip) const;
    int accessibility_request_count() const;
};

enum class Role { Panel, Label, Button, Checkbox, Progress, Text, Table, Row, ScrollBar };

struct MouseEvent { Type {Move, Down, Up, Wheel, Leave} type; float x, y; int wheel_delta; bool left_button; };
struct KeyEvent   { UINT vk; bool ctrl, shift, alt; bool is_down; };

class Widget {
    // 字段
    RectF bounds; int control_id; Widget* parent; Host* host;
    bool visible, enabled, tab_stop, focused, focus_visible, hovered, pressed;
    std::wstring text;

    // 生命周期与绘制
    virtual void paint(Canvas&) = 0;
    virtual Widget* hit_test(float x, float y);
    virtual bool on_mouse(const MouseEvent&);
    virtual bool on_key(const KeyEvent&);
    virtual void tick(float dt_seconds);
    virtual void on_focus();
    virtual void on_blur();
    virtual void on_theme_changed();

    // 无障碍
    virtual std::wstring accessible_name() const;      // 默认 text
    virtual Role accessible_role() const = 0;
    virtual std::wstring accessible_value() const;
    virtual std::wstring accessible_default_action() const;
    virtual void do_default_action();
    virtual uint32_t accessible_state() const;
    virtual int accessible_row_count() const;          // Table 覆盖
    virtual bool is_default_button() const;

    virtual const std::vector<std::unique_ptr<Widget>>* child_widgets() const;

    bool accepts_focus() const;      // tab_stop && visible && enabled
    void invalidate();
    void invalidate_with(const RectF&);
    void focus();
    const Theme& theme() const;

protected:
    void paint_focus_ring(Canvas&, float radius = 6.0f) const;
};

class Panel : public Widget {
    Widget* add(std::unique_ptr<Widget> child);

};
